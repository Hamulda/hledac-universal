1//! Aho-Corasick multi-pattern matcher implementation.
2
3use pyo3::prelude::*;
4use aho_corasick::AhoCorasick;
5
6/// Aho-Corasick multi-pattern matcher for fast IOC detection.
7///
8/// # Example
9/// ```python
10/// from hledac_rust_extensions import AhoCorasickMatcher
11/// matcher = AhoCorasickMatcher(patterns=["malware", "phishing", "suspicious"])
12/// results = matcher.scan("Check out this phishing site")
13/// # Returns: [(start, end, pattern), ...]
14/// ```
15#[pyclass]
16pub struct AhoCorasickMatcher {
17    automaton: AhoCorasick,
18    patterns: Vec<String>,
19}
20
21#[pymethods]
22impl AhoCorasickMatcher {
23    /// Create a new AhoCorasickMatcher with the given patterns.
24    #[new]
25    #[pyo3(signature = (patterns = vec![]))]
26    fn new(patterns: Vec<String>) -> Self {
27        let automaton = AhoCorasick::new(&patterns).expect("Failed to build automaton");
28        Self { automaton, patterns }
29    }
30
31    /// Scan text and return all pattern matches.
32    ///
33    /// Returns list of (start, end, pattern_name) tuples.
34    fn scan(&self, text: &str) -> Vec<(usize, usize, String)> {
35        let mut results = Vec::new();
36        for m in self.automaton.find_iter(text.as_bytes()) {
37            if let Some(pattern) = self.patterns.get(m.pattern().as_usize()) {
38                let start = m.start();
39                let end = m.end();
40                results.push((start, end, pattern.clone()));
41            }
42        }
43        results
44    }
45
46    /// Get the number of patterns.
47    fn len(&self) -> usize {
48        self.patterns.len()
49    }
50
51    /// Check if no patterns are loaded.
52    fn is_empty(&self) -> bool {
53        self.patterns.is_empty()
54    }
55
56    /// Fast path: return True if any pattern matches, False otherwise.
57    /// Optimized for short-circuit evaluation on large texts.
58    fn find_any(&self, text: &str) -> bool {
59        self.automaton.is_match(text.as_bytes())
60    }
61}
