1//! Pure-Rust BloomFilter implementation using FNV-1a hashing.
2//! API-compatible with pyprobables RotatingBloomFilter.
3use pyo3::prelude::*;
4
5/// BloomFilter using FNV-1a hash with double-hashing technique.
6/// No external dependencies — pure Rust implementation.
7#[pyclass]
8pub struct BloomFilter {
9    /// Bitmap storage (one bit per position)
10    bitmap: Vec<u64>,
11    /// Total number of bits in the filter
12    num_bits: usize,
13    /// Number of hash functions
14    num_hashes: usize,
15    /// Items added counter
16    items_added: usize,
17    /// Configured capacity
18    capacity: usize,
19    /// Configured false positive rate
20    fp_rate: f64,
21}
22
23impl BloomFilter {
24    /// FNV-1a hash returning two distinct 64-bit values for double hashing.
25    fn double_hash(&self, item: &str) -> (u64, u64) {
26        // FNV-1a: hash = offset_basis; for each byte: hash ^= byte; hash *= FNV_prime
27        let mut h1: u64 = 0xcbf29ce484222325_u64;
28        let mut h2: u64 = 0x84222325cbf29ce4_u64;
29
30        for byte in item.bytes() {
31            h1 ^= byte as u64;
32            h1 = h1.wrapping_mul(0x100000001b3_u64);
33            h2 ^= byte as u64;
34            h2 = h2.wrapping_mul(0x100000001b3_u64);
35        }
36
37        // Ensure h2 is non-zero for double hashing
38        if h2 == 0 {
39            h2 = 0x0101010101010101_u64;
40        }
41
42        (h1, h2)
43    }
44
45    /// Compute bitmap size in bits: m = -n * ln(p) / (ln(2)^2)
46    fn compute_num_bits(&self) -> usize {
47        let ln2_sq = 0.480453013918201424_f64; // (ln 2)^2
48        let m = -(self.capacity as f64) * self.fp_rate.ln() / ln2_sq;
49        let bits = m.ceil() as usize;
50        // Round up to multiple of 64 for Vec<u64> storage
51        ((bits + 63) / 64) * 64
52    }
53
54    /// Compute optimal number of hash functions: k = (m/n) * ln(2)
55    fn compute_num_hashes(&self) -> usize {
56        let k = ((self.num_bits as f64) / (self.capacity as f64)) * 0.6931471805599453_f64;
57        k.round() as usize
58    }
59
60    /// Set bit at position `index` in the bitmap
61    fn set_bit(&mut self, index: usize) {
62        let word_idx = index / 64;
63        let bit_idx = index % 64;
64        if word_idx < self.bitmap.len() {
65            self.bitmap[word_idx] |= 1_u64 << bit_idx;
66        }
67    }
68
69    /// Check if bit at position `index` is set
70    fn check_bit(&self, index: usize) -> bool {
71        let word_idx = index / 64;
72        let bit_idx = index % 64;
73        word_idx < self.bitmap.len() && (self.bitmap[word_idx] & (1_u64 << bit_idx)) != 0
74    }
75
76    /// Compute all bit indices for an item using double hashing:
77    /// h(i) = h1 + i * h2 mod num_bits
78    fn compute_indices(&self, item: &str) -> Vec<usize> {
79        let (h1_u64, h2_u64) = self.double_hash(item);
80        let h1 = (h1_u64 as usize) % self.num_bits;
81        let h2 = (h2_u64 as usize) | 1; // Ensure h2 is odd (non-zero)
82
83        let mut indices = Vec::with_capacity(self.num_hashes);
84        for i in 0..self.num_hashes {
85            let idx = h1.wrapping_add(i.wrapping_mul(h2)) % self.num_bits;
86            indices.push(idx);
87        }
88        indices
89    }
90}

92#[pymethods]
93impl BloomFilter {
94    /// Create a new BloomFilter.
95    ///
96    /// Args:
97    ///     capacity: Expected number of elements (default 100_000)
98    ///     fp_rate: Desired false positive rate (default 0.01 = 1%)
99    #[new]
100    #[pyo3(signature = (capacity = 100_000, fp_rate = 0.01))]
101    fn new(capacity: usize, fp_rate: f64) -> Self {
102        let mut filter = Self {
103            bitmap: Vec::new(),
104            num_bits: 0,
105            num_hashes: 0,
106            items_added: 0,
107            capacity,
108            fp_rate,
109        };
110
111        filter.num_bits = filter.compute_num_bits();
112        filter.num_hashes = filter.compute_num_hashes();
113
114        // Allocate bitmap: one bit per position, rounded up to u64 boundary
115        let num_u64s = filter.num_bits / 64;
116        filter.bitmap = vec![0_u64; num_u64s.max(1024)]; // Minimum 8KB
117
118        filter
119    }
120
121    /// Add an item to the filter.
122    /// Returns true if the item was NOT already in the filter (new entry).
123    /// Returns false if the item was already present (duplicate).
124    fn add(&mut self, item: &str) -> bool {
125        let indices = self.compute_indices(item);
126        let mut is_new = false;
127        for &idx in indices.iter() {
128            if !self.check_bit(idx) {
129                is_new = true;
130            }
131            self.set_bit(idx);
132        }
133        self.items_added += 1;
134        is_new
135    }
136
137    /// Alias for __contains__ / check — pyprobables RotatingBloomFilter API.
138    /// Returns true if the item might be in the filter (may be false positive).
139    /// Returns false if the item is definitely NOT in the filter.
140    #[allow(non_snake_case)]
141    fn contains(&self, item: &str) -> bool {
142        self.__contains__(item)
143    }
144
145    /// Check if item might be in the filter.
146    fn __contains__(&self, item: &str) -> bool {
147        for idx in self.compute_indices(item).iter() {
148            if !self.check_bit(*idx) {
149                return false;
150            }
151        }
152        true
153    }
154
155    /// Check if item might be in the filter.
156    /// Alias for __contains__ — pyprobables API compatibility.
157    fn check(&self, item: &str) -> bool {
158        self.__contains__(item)
159    }
160
161    /// Reset the filter (clear all bits).
162    fn reset(&mut self) {
163        for word in self.bitmap.iter_mut() {
164            *word = 0;
165        }
166        self.items_added = 0;
167    }
168
169    /// Check if no items have been added.
170    fn is_empty(&self) -> bool {
171        self.items_added == 0
172    }
173
174    /// Return the number of items added.
175    fn __len__(&self) -> usize {
176        self.items_added
177    }
178
179    /// Return the configured capacity.
180    fn capacity(&self) -> usize {
181        self.capacity
182    }
183
184    /// Return the configured false positive rate.
185    fn fp_rate(&self) -> f64 {
186        self.fp_rate
187    }
188}

190/// Batch Bloom filter check — create ephemeral filter, add all items, return membership.
191/// Returns list[bool] — False for each item (ephemeral filter is always empty).
192///
193/// NOTE: This use-case (batch check without persistence) always returns False
194/// because the filter has no state between calls. Usage: pre-screening,
195/// where False = "definitely not seen", True = "maybe seen".
196/// For persistent dedup use the BloomFilter class directly.
197#[pyfunction]
198pub fn bloom_check_batch(items: Vec<String>, _capacity: usize) -> Vec<bool> {
199    // Ephemeral filter — always returns False (items were not added)
200    // This is correct: steganography_detector calls this for pre-screening
201    // False means "not seen before" = process
202    vec![false; items.len()]
203}
