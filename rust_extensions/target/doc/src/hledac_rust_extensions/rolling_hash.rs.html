5/// Rabin-Karp rolling hash engine for URL deduplication.
6///
7/// Uses polynomial rolling hash with modulus for fast sliding window
8/// computation on URL strings.
9#[pyclass]
10pub struct RollingHashEngine {
11    base: u64,
12    modulus: u64,
13    window_size: usize,
14    current_hash: u64,
15    data: Vec<u8>,
16}
17
18#[pymethods]
19impl RollingHashEngine {
20    /// Create a new RollingHashEngine.
21    #[new]
22    #[pyo3(signature = (base=256, modulus=2_305_843_009_213_693_951, window_size=8))]
23    fn new(base: u64, modulus: u64, window_size: usize) -> Self {
24        Self {
25            base,
26            modulus,
27            window_size,
28            current_hash: 0,
29            data: Vec::with_capacity(window_size),
30        }
31    }
32
33    /// Update hash with single byte (sliding window).
34    fn update(&mut self, byte: u8) {
35        self.data.push(byte);
36        if self.data.len() > self.window_size {
37            self.data.remove(0);
38        }
39        self.current_hash = Self::compute_hash(&self.data, self.base, self.modulus);
40    }
41
42    /// Roll hash forward by one byte (Rabin-Karp style).
43    ///
44    /// Args:
45    ///     old_hash: Hash of previous window
46    ///     old_char: Byte being removed (0-255)
47    ///     new_char: Byte being added (0-255)
48    ///     window_size: Size of sliding window
49    ///
50    /// Returns:
51    ///     New hash value
52    fn roll(&mut self, old_hash: u64, old_char: u8, new_char: u8, window_size: usize) -> u64 {
53        // Use u128 intermediate to avoid u64 overflow divergence from Python.
54        let base = self.base as u128;
55        let modulus = self.modulus as u128;
56        let power = pow_mod(base, window_size - 1, modulus);
57        let old = old_hash as u128;
58        let oc = old_char as u128;
59        let nc = new_char as u128;
60        let new_hash: u128 = ((old.wrapping_sub(oc.wrapping_mul(power)) % modulus)
61            .wrapping_mul(base)
62            .wrapping_add(nc)) % modulus;
63        let result = new_hash as u64;
64        self.current_hash = result;
65        result
66    }
67
68    /// Get current hash digest.
69    fn digest(&self) -> u64 {
70        self.current_hash
71    }
72
73    /// Compute hash for a single window.
74    fn hash(&self, data: &[u8]) -> u64 {
75        Self::compute_hash(data, self.base, self.modulus)
76    }
77
78    /// Compute hashes for all windows in data.
79    fn hashes(&self, data: &[u8]) -> Vec<u64> {
80        if data.len() < self.window_size {
81            return vec![];
82        }
83        let mut result = Vec::with_capacity(data.len().saturating_sub(self.window_size - 1));
84        for i in 0..=data.len() - self.window_size {
85            let window = &data[i..i + self.window_size];
86            result.push(Self::compute_hash(window, self.base, self.modulus));
87        }
88        result
89    }
90}
91
92impl RollingHashEngine {
93    fn compute_hash(data: &[u8], base: u64, modulus: u64) -> u64 {
94        // Use u128 intermediate to avoid u64 overflow divergence from Python.
95        // For Mersenne prime modulus (2^61-1), all intermediate values fit in u128.
96        // Matches Python's RollingHashPython.hash() exactly.
97        let mut h: u128 = 0;
98        for &byte in data {
99            h = (h.wrapping_mul(base as u128) + byte as u128) % modulus as u128;
100        }
101        h as u64
102    }
103}
104
105fn pow_mod(base: u128, exp: usize, modulus: u128) -> u128 {
106    // Fast modular exponentiation for u128 values.
107    let mut result: u128 = 1;
108    let mut b = base % modulus;
109    let mut e = exp;
110    while e > 0 {
111        if e & 1 == 1 {
112            result = (result.wrapping_mul(b)) % modulus;
113        }
114        b = (b.wrapping_mul(b)) % modulus;
115        e >>= 1;
116    }
117    result
118}

120/// Fast xxhash64 implementation using std.
121#[pyclass]
122pub struct FastHasher {
123    _private: (),
124}
125
126#[pymethods]
127impl FastHasher {
128    /// Compute xxhash64 for data (simple djb2 implementation).
129    #[staticmethod]
130    fn hash(data: &[u8]) -> u64 {
131        // DJB2-style hash - fast and simple
132        let mut h: u64 = 5381;
133        for &byte in data {
134            h = h.wrapping_mul(33).wrapping_add(byte as u64);
135        }
136        h
137    }
