1️⃣1️⃣ Thread-Safe State Management RLock

Location: transport/circuit_breaker.py lines 217-254

@dataclasses.dataclass
class CircuitBreaker:
    domain: str
    failure_threshold: int = CIRCUIT_FAILURE_THRESHOLD
    recovery_timeout: float = BASE_RECOVERY_TIMEOUT_S
    _state_lock: threading.RLock = dataclasses.field(default=None, init=False)

    def __post_init__(self) -> None:
        if self._state_lock is None:
            self._state_lock = threading.RLock()  # Reentrant for callbacks

    def is_open(self) -> bool:
        with self._state_lock:
            if self._state == CBState.OPEN:
                if time.monotonic() - self._last_failure_time > self.recovery_timeout:
                    self._transition_to_half_open()
                    return False  # HALF_OPEN allows one probe
                return True  # Still open, fail fast
            return False  # CLOSED, proceed

1️⃣2️⃣ Full Jitter Backoff F285-JITTER

Location: transport/circuit_breaker.py lines 290-305

def _jittered_retry_after(self) -> float:
    """F285-JITTER: Compute jittered retry_after in [0.5*timeout, 1.5*timeout].
    
    Full jitter prevents thundering herd when many requests wake up
    simultaneously after recovery_timeout. Each caller independently
    samples from the same range, spreading out retry attempts.
    """
    try:
        raw = _JITTER_RNG.uniform(
            _JITTER_MIN_MULTIPLIER * self.recovery_timeout,
            _JITTER_MAX_MULTIPLIER * self.recovery_timeout,
        )
        return max(0.0, raw)
    except Exception:
        return self.recovery_timeout  # Fallback

1️⃣4️⃣ TCP Keep-Alive Patching ISSUE-P6-001

Location: transport/session_pool.py lines 71-151

def _patch_socket_keepalive(sock: socket.socket) -> None:
    """ISSUE-P6-001: Patch TCP keep-alive options.
    
    Enables kernel-level TCP keep-alive probes so dead connections are detected
    proactively (before TIME_WAIT timeout), freeing pool slots earlier.
    
    macOS: TCP_KEEPIDLE=0x10 (60s), TCP_KEEPINTVL=0x101 (30s), TCP_KEEPCNT=0x102 (3)
    """
    sock.setsockopt(socket.SOL_SOCKET, _SO_KEEPALIVE, 1)
    sock.setsockopt(socket.IPPROTO_TCP, _TCP_KEEPIDLE, _TCP_KEEPALIVE_IDLE_S)
    sock.setsockopt(socket.IPPROTO_TCP, _TCP_KEEPINTVL, _TCP_KEEPALIVE_INTERVAL_S)
    sock.setsockopt(socket.IPPROTO_TCP, _TCP_KEEPCNT, _TCP_KEEPALIVE_MAX_PROBES)

async def _probe_http2_negotiation(client: httpx.AsyncClient) -> bool:
    """ISSUE-P6-002: Probe whether HTTP/2 was actually negotiated.
    
    Detects silent HTTP/1.1 fallback when h2 is installed but ALPN fails.
    Uses lightweight HEAD to Cloudflare to verify negotiation.
    """
    probe_url = "https://www.cloudflare.com/cdn-cgi/trace"
    resp = await client.get(probe_url, timeout=httpx.Timeout(connect=3.0, read=3.0))
    http_version = resp.extensions.get("http_version", "")
    if http_version and http_version.lower() in ("h2", "http/2", "2"):
        _http2_negotiated = True
    else:
        _http2_negotiated = False  # Silent fallback detected
