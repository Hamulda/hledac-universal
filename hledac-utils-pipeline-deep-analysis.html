PEP 562 __getattr__ lazy loading — submodules se načítají až při prvním

runtime.sprint_scheduler.
"""

_SUBMODULE_REGISTRY: dict[str, tuple[str, tuple[str, ...]]] = {
    "action_result": (".action_result", ("ActionResult",)),
    "async_utils": (".async_utils", ("TaskResult", "bounded_map", "map_as_completed")),
    "bloom_filter": (".bloom_filter", ("BloomFilter", ...)),
    "deduplication": (".deduplication", ("ContentDeduplicator", ...)),
    "intelligent_cache": (".intelligent_cache", ("CacheConfig", ...)),
    # ... 89 total submodules
}

def __getattr__(name: str) -> object | None:
    """Lazy import on first access."""
    if name in _SUBMODULE_REGISTRY:
        module_path, exports = _SUBMODULE_REGISTRY[name]
        mod = importlib.import_module(module_path, __package__)
        return functools.reduce(getattr, exports, mod)
    raise AttributeError(f"module {__name__!r} has no attribute {name!r}")

Benefit: Sprint scheduler imports only what it needs, not all 89 modules. ~3300ms saved.

8️⃣ AIMD Flow Control P2-3

Location: pipeline/_stage_protocol.py

class AIMDController:
    """Additive Increase, Multiplicative Decrease flow control.
    
    Used between pipeline stages to prevent queue overflow.
    
    Lifecycle:
    - AI: Increment window by 1 per successful batch
    - MD: Multiply window by 0.5 on overflow/drop
    - Floor: Never go below min_window (typically 1)
    """
    
    def __init__(self, initial: int = 10, min_window: int = 1, max_window: int = 100):
        self.window = initial
        self.min_window = min_window
        self.max_window = max_window
    
    def on_success(self) -> None:
        self.window = min(self.window + 1, self.max_window)
    
    def on_overflow(self) -> None:
        self.window = max(int(self.window * 0.5), self.min_window)
