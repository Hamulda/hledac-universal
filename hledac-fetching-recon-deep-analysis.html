1️⃣ Factory Pattern - Lazy-Loaded Resources

psutil & Content Hasher Factories Thread-Safe

Location: fetching/_factories.py

def make_psutil_getter():
    """Factory: Creates a lazy psutil getter with cached import."""
    _psutil: Any = None
    def getter() -> Any:
        nonlocal _psutil
        if _psutil is not None:
            return _psutil
        try:
            import psutil as _ps
            _psutil = _ps
        except Exception:
            _psutil = None
        return _psutil
    return getter

def make_content_hasher_factory():
    """Factory: Creates lazy content hasher with Rust backend fallback."""
    _ContentHasher: Any = None
    _RUST_CONTENT_HASHER: bool = False
    def getter() -> Any:
        nonlocal _ContentHasher, _RUST_CONTENT_HASHER
        if _RUST_CONTENT_HASHER:
            return _ContentHasher
        try:
            from hledac.universal.core.rust_backend import rust
            _ContentHasher = rust.hash
            _RUST_CONTENT_HASHER = True
        except Exception:
            _RUST_CONTENT_HASHER = False
            _ContentHasher = None
        return _ContentHasher
    return getter, is_rust

Benefit: Thread-safe, fail-soft, lazy import. Rust backend preferred, xxHash fallback.

class BrowserDecision(msgspec.Struct, frozen=True, gc=False):
    tier: BrowserTier  # "camoufox", "nodriver", "deferred", "skip_js"
    allowed: bool
    rss_gib: float
    js_confidence: float
    reason: str

def decide(*, js_confidence: float, priority: int = 5) -> BrowserDecision:
    """Pure runtime decision. No env flags. No config. Called per-request."""
    rss = _rss_gib()
    if rss >= _HARD_GIB:
        return BrowserDecision(tier="deferred", allowed=False, ...)
    if rss >= _SOFT_GIB:
        if priority <= 3 and js_confidence >= 0.75:
            return BrowserDecision(tier="camoufox", allowed=True, ...)
        return BrowserDecision(tier="deferred", allowed=False, ...)
    return BrowserDecision(tier="camoufox", allowed=True, ...)
RSS < 4.5 GiB → camoufox allowed
RSS 4.5-6.0 GiB → camoufox only if priority≤3 AND confidence≥0.75
RSS ≥ 6.0 GiB → deferred always
