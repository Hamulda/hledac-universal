def _compute_reward(latency_ms, status_code, got_captcha) -> float:
    # Speed reward: linear decay 1.0 (0ms) → 0.0 (5000ms)
    speed_reward = max(0.0, 1.0 - latency_ms / 5000.0)
    
    # Detection penalty
    if status_code == 429:  penalty = 0.8   # Rate limit
    elif status_code == 403: penalty = 0.6   # Forbidden
    elif got_captcha:       penalty = 0.9   # CAPTCHA
    else:                   penalty = 0.0
    
    return speed_reward * (1.0 - penalty)

class RateLimiter:
    """ISSUE 24: Uses Rust try_acquire() via asyncio.to_thread()
       ~10× faster than asyncio.Lock polling."""
    
    async def acquire(self) -> None:
        if self._rust_limiter is not None:
            # Rust path: non-blocking try_acquire in thread pool
            while True:
                acquired = await asyncio.to_thread(
                    self._rust_limiter.try_acquire
                )
                if acquired:
                    return
                await asyncio.sleep(0.1)  # Cooperative sleep
        
        # Python fallback: asyncio.Lock + refill + sleep
